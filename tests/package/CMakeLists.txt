cmake_minimum_required(VERSION 3.24)
project(iiCameraPackageContract LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_iiGraphicAdjust TRUE)
if(MODE STREQUAL "Core")
    find_package(iiCameraIntergration 0.2 CONFIG REQUIRED COMPONENTS Core)
elseif(MODE STREQUAL "Optional")
    find_package(iiCameraIntergration 0.2 CONFIG REQUIRED COMPONENTS Core OPTIONAL_COMPONENTS ImageProcessing)
    if(iiCameraIntergration_ImageProcessing_FOUND OR TARGET iiCameraIntergration::ImageProcessing)
        message(FATAL_ERROR "Unavailable optional image dependency was exposed")
    endif()
elseif(MODE STREQUAL "Required")
    find_package(iiCameraIntergration 0.2 CONFIG REQUIRED COMPONENTS Core ImageProcessing)
else()
    message(FATAL_ERROR "Unknown package contract mode")
endif()
if(NOT TARGET iiCameraIntergration::iiCameraIntergration)
    message(FATAL_ERROR "Core target is missing")
endif()
