cmake_minimum_required(VERSION 3.25)
project(iiCameraIntergrationInstalledConsumer LANGUAGES CXX)

option(IICAMERA_CONSUMER_WITH_PROCESSING "Test the optional processing component" OFF)
if(IICAMERA_CONSUMER_WITH_PROCESSING)
    find_package(iiCameraIntergration 0.2 CONFIG REQUIRED COMPONENTS Core ImageProcessing)
else()
    find_package(iiCameraIntergration 0.2 CONFIG REQUIRED COMPONENTS Core)
endif()
add_executable(iiCameraIntergration_consumer main.cpp)
target_compile_features(iiCameraIntergration_consumer PRIVATE cxx_std_23)
if(IICAMERA_CONSUMER_WITH_PROCESSING)
    target_link_libraries(iiCameraIntergration_consumer PRIVATE iiCameraIntergration::ImageProcessing)
    target_compile_definitions(iiCameraIntergration_consumer PRIVATE IICAMERA_CONSUMER_WITH_PROCESSING)
else()
    target_link_libraries(iiCameraIntergration_consumer PRIVATE iiCameraIntergration::iiCameraIntergration)
endif()
enable_testing()
add_test(NAME iiCameraIntergration.installed_consumer COMMAND iiCameraIntergration_consumer)
