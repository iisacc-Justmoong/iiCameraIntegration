cmake_minimum_required(VERSION 3.25)
project(iiCameraIntegrationInstalledConsumer LANGUAGES CXX)

option(IICAMERA_CONSUMER_WITH_PROCESSING "Test the optional processing component" OFF)
if(IICAMERA_CONSUMER_WITH_PROCESSING)
    find_package(iiCameraImageProcessing 0.4 CONFIG REQUIRED)
else()
    find_package(iiCameraIntegration 0.4 CONFIG REQUIRED COMPONENTS Core)
endif()
add_executable(iiCameraIntegration_consumer main.cpp)
target_compile_features(iiCameraIntegration_consumer PRIVATE cxx_std_23)
if(IICAMERA_CONSUMER_WITH_PROCESSING)
    target_link_libraries(iiCameraIntegration_consumer PRIVATE iiCameraIntegration::ImageProcessing)
    target_compile_definitions(iiCameraIntegration_consumer PRIVATE IICAMERA_CONSUMER_WITH_PROCESSING)
else()
    target_link_libraries(iiCameraIntegration_consumer PRIVATE iiCameraIntegration::iiCameraIntegration)
endif()
enable_testing()
add_test(NAME iiCameraIntegration.installed_consumer COMMAND iiCameraIntegration_consumer)
