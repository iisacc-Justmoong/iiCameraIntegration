cmake_minimum_required(VERSION 3.24)
project(iiCameraIntegration VERSION 0.4.1 LANGUAGES CXX)




set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(GenerateExportHeader)
include(CTest)
option(BUILD_SHARED_LIBS "Build a shared iiCameraIntegration library" ON)
option(IICAMERA_WITH_FFMPEG "Enable FFmpeg codec/format recognition and capture" ON)
option(IICAMERA_WITH_NATIVE "Enable the platform native camera metadata/control adapter" ON)
option(IICAMERA_WITH_IMAGE_PROCESSING "Deprecated: configure example/ImageProcessing independently" OFF)
if(IICAMERA_WITH_IMAGE_PROCESSING)
    message(FATAL_ERROR "Configure example/ImageProcessing as a separate application consumer")
endif()
option(IICAMERA_BUILD_TOOLS "Deprecated: configure example/Media independently" OFF)
if(IICAMERA_BUILD_TOOLS)
    message(FATAL_ERROR "Configure example/Media as a separate application consumer")
endif()
if(PROJECT_IS_TOP_LEVEL AND CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    if(DEFINED ENV{HOME} AND NOT "$ENV{HOME}" STREQUAL "")
        set(CMAKE_INSTALL_PREFIX "$ENV{HOME}/.local/SDK/iiCameraIntegration" CACHE PATH "Install prefix" FORCE)
    elseif(DEFINED ENV{USERPROFILE} AND NOT "$ENV{USERPROFILE}" STREQUAL "")
        set(CMAKE_INSTALL_PREFIX "$ENV{USERPROFILE}/.local/SDK/iiCameraIntegration" CACHE PATH "Install prefix" FORCE)
    endif()
endif()
add_library(iiCameraIntegration src/iiCameraIntegration/Camera.cpp src/iiCameraIntegration/Devices.cpp
    src/iiCameraIntegration/Features.cpp)
add_library(iiCameraIntegration::iiCameraIntegration ALIAS iiCameraIntegration)
target_compile_features(iiCameraIntegration PUBLIC cxx_std_23)
target_include_directories(iiCameraIntegration PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
generate_export_header(iiCameraIntegration BASE_NAME IICAMERAINTEGRATION
    EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/include/iiCameraIntegration/Export.h")
if(NOT BUILD_SHARED_LIBS)
    target_compile_definitions(iiCameraIntegration PUBLIC IICAMERAINTEGRATION_STATIC_DEFINE)
endif()
set_target_properties(iiCameraIntegration PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR})
if(MSVC)
    target_compile_options(iiCameraIntegration PRIVATE /W4 /permissive-)
else()
    target_compile_options(iiCameraIntegration PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(iiCameraIntegration PRIVATE Threads::Threads)
if(IICAMERA_WITH_FFMPEG)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(IICameraFFmpeg REQUIRED IMPORTED_TARGET
        libavformat libavcodec libavdevice libavutil libswscale)
    target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/Media.cpp)
    target_link_libraries(iiCameraIntegration PRIVATE PkgConfig::IICameraFFmpeg)
else()
    target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/MediaUnavailable.cpp)
endif()
set(IICAMERA_HAS_APPLE_NATIVE OFF)
if(APPLE AND IICAMERA_WITH_NATIVE)
    enable_language(OBJCXX)
    set(IICAMERA_HAS_APPLE_NATIVE ON)
    target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/NativeCamera.mm)
    set_source_files_properties(src/iiCameraIntegration/NativeCamera.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    set_target_properties(iiCameraIntegration PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS OFF)
    find_library(IICameraAVFoundation AVFoundation REQUIRED)
    find_library(IICameraFoundation Foundation REQUIRED)
    find_library(IICameraCoreMedia CoreMedia REQUIRED)
    find_library(IICameraCoreVideo CoreVideo REQUIRED)
    target_link_libraries(iiCameraIntegration PRIVATE "${IICameraAVFoundation}" "${IICameraFoundation}" "${IICameraCoreMedia}" "${IICameraCoreVideo}")
elseif(ANDROID AND IICAMERA_WITH_NATIVE)
    if(CMAKE_SYSTEM_VERSION VERSION_LESS 24)
        message(STATUS "Native Camera2 unavailable before Android API 24; building the explicit fallback")
        target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/NativeCameraUnavailable.cpp)
    else()
        target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/NativeCameraAndroid.cpp)
        target_link_libraries(iiCameraIntegration PRIVATE camera2ndk)
    endif()
elseif(WIN32 AND IICAMERA_WITH_NATIVE)
    target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/NativeCameraWindows.cpp)
    target_link_libraries(iiCameraIntegration PRIVATE mfplat mf mfuuid strmiids ole32 oleaut32 propsys)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND IICAMERA_WITH_NATIVE)
    target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/NativeCameraLinux.cpp
        src/iiCameraIntegration/V4L2.cpp)
else()
    target_sources(iiCameraIntegration PRIVATE src/iiCameraIntegration/NativeCameraUnavailable.cpp)
endif()
set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiCameraIntegration")
install(TARGETS iiCameraIntegration EXPORT iiCameraIntegrationTargets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/iiCameraIntegration/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiCameraIntegration FILES_MATCHING PATTERN "*.h" PATTERN "ImageProcessing.h" EXCLUDE PATTERN "detail" EXCLUDE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/iiCameraIntegration/Export.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiCameraIntegration)
install(EXPORT iiCameraIntegrationTargets NAMESPACE iiCameraIntegration:: DESTINATION "${_package_dir}")
configure_package_config_file(cmake/iiCameraIntegrationConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntegrationConfig.cmake" INSTALL_DESTINATION "${_package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntegrationConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntegrationConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntegrationConfigVersion.cmake" DESTINATION "${_package_dir}")
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCameraIntegration)
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCameraIntegration/docs
    PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
install(DIRECTORY example/ DESTINATION ${CMAKE_INSTALL_DATADIR}/iiCameraIntegration/example
    PATTERN ".*.iisacc-preview" EXCLUDE PATTERN ".iisacc-temporary.lock" EXCLUDE PATTERN ".DS_Store" EXCLUDE)
if(BUILD_TESTING)
    add_test(NAME iiCamera.graphic_sdk_boundary COMMAND "${CMAKE_COMMAND}"
        "-DIICAMERA_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/graphic_sdk_boundary.cmake")
    add_test(NAME iiCamera.provider_sdk_boundary COMMAND "${CMAKE_COMMAND}"
        "-DIICAMERA_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/provider_sdk_boundary.cmake")
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME iiCamera.installed_package COMMAND "${CMAKE_COMMAND}"
            "-DBUILD=${CMAKE_CURRENT_BINARY_DIR}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
            "-DCONFIG=$<CONFIG>" "-DLIBDIR=${CMAKE_INSTALL_LIBDIR}"
            "-DGENERATOR=${CMAKE_GENERATOR}" "-DCXX_COMPILER=${CMAKE_CXX_COMPILER}"
            "-DDEPENDENCY_PREFIX_PATH=${CMAKE_PREFIX_PATH}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/installed_package.cmake")
        set_tests_properties(iiCamera.installed_package PROPERTIES TIMEOUT 180)
    endif()
    set(IICAMERA_ANDROID_TEST_INCLUDE_DIR "" CACHE PATH "Optional official NDK headers for artificial Camera2 tests on a non-Android host")
    if(IICAMERA_ANDROID_TEST_INCLUDE_DIR AND NOT ANDROID)
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/android-test-api.h"
            "#define __INTRODUCED_IN(x)\n#define __INTRODUCED_IN_32(x)\n#define __INTRODUCED_IN_64(x)\n#define __ANDROID_API__ 24\n")
        add_executable(iiCameraAndroidTests tests/android.cpp src/iiCameraIntegration/NativeCameraAndroid.cpp)
        target_link_libraries(iiCameraAndroidTests PRIVATE iiCameraIntegration::iiCameraIntegration)
        target_compile_definitions(iiCameraAndroidTests PRIVATE NativeCameraSystem=AndroidFixtureSystem)
        target_compile_options(iiCameraAndroidTests PRIVATE -include "${CMAKE_CURRENT_BINARY_DIR}/android-test-api.h")
        target_include_directories(iiCameraAndroidTests PRIVATE "${IICAMERA_ANDROID_TEST_INCLUDE_DIR}")
        add_test(NAME iiCamera.android_driver COMMAND iiCameraAndroidTests)
        set_tests_properties(iiCamera.android_driver PROPERTIES TIMEOUT 30)
    endif()
    set(IICAMERA_V4L2_TEST_INCLUDE_DIR "" CACHE PATH "Optional Linux UAPI headers for artificial V4L2 driver tests on a non-Linux host")
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR IICAMERA_V4L2_TEST_INCLUDE_DIR)
        add_executable(iiCameraV4L2Tests tests/v4l2.cpp src/iiCameraIntegration/V4L2.cpp)
        target_link_libraries(iiCameraV4L2Tests PRIVATE iiCameraIntegration::iiCameraIntegration)
        if(IICAMERA_V4L2_TEST_INCLUDE_DIR)
            target_include_directories(iiCameraV4L2Tests PRIVATE "${IICAMERA_V4L2_TEST_INCLUDE_DIR}")
        endif()
        add_test(NAME iiCamera.v4l2_driver COMMAND iiCameraV4L2Tests)
        set_tests_properties(iiCamera.v4l2_driver PROPERTIES TIMEOUT 30)
    endif()
    add_executable(iiCameraFeatureJsonTests tests/features_json.cpp)
    target_link_libraries(iiCameraFeatureJsonTests PRIVATE iiCameraIntegration::iiCameraIntegration)
    add_test(NAME iiCamera.features.json COMMAND "${CMAKE_COMMAND}"
        "-DTOOL=$<TARGET_FILE:iiCameraFeatureJsonTests>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/features_json.cmake")
    add_executable(iiCameraFeatureTests tests/features.cpp)
    target_link_libraries(iiCameraFeatureTests PRIVATE iiCameraIntegration::iiCameraIntegration)
    foreach(_case bulk devices lifecycle concurrency)
        add_test(NAME iiCamera.features.${_case} COMMAND iiCameraFeatureTests ${_case})
        set_tests_properties(iiCamera.features.${_case} PROPERTIES TIMEOUT 30)
    endforeach()
    add_executable(iiCameraDeviceTests tests/devices.cpp)
    target_link_libraries(iiCameraDeviceTests PRIVATE iiCameraIntegration::iiCameraIntegration Threads::Threads)
    foreach(_case discovery pairing lifecycle controls failures concurrency validation)
        add_test(NAME iiCamera.devices.${_case} COMMAND iiCameraDeviceTests ${_case})
        set_tests_properties(iiCamera.devices.${_case} PROPERTIES TIMEOUT 30)
    endforeach()
    add_executable(iiCameraTests tests/camera.cpp)
    target_link_libraries(iiCameraTests PRIVATE iiCameraIntegration::iiCameraIntegration)
    foreach(_case metadata_preservation metadata_validation numeric_ranges control_preflight provider_contract optics)
        add_test(NAME iiCamera.${_case} COMMAND iiCameraTests ${_case})
        set_tests_properties(iiCamera.${_case} PROPERTIES TIMEOUT 10)
    endforeach()
    add_executable(iiCameraNativeTests tests/native.cpp)
    target_link_libraries(iiCameraNativeTests PRIVATE iiCameraIntegration::iiCameraIntegration)
    add_test(NAME iiCamera.native COMMAND iiCameraNativeTests)
    set_tests_properties(iiCamera.native PROPERTIES TIMEOUT 20)
    if(NOT IICAMERA_WITH_FFMPEG)
        add_executable(iiCameraUnavailableTests tests/unavailable.cpp)
        target_link_libraries(iiCameraUnavailableTests PRIVATE iiCameraIntegration::iiCameraIntegration)
        add_test(NAME iiCamera.unavailable COMMAND iiCameraUnavailableTests)
    endif()
    if(IICAMERA_WITH_FFMPEG)
        add_executable(iiCameraColorDecodeTests tests/color_decode.cpp)
        target_link_libraries(iiCameraColorDecodeTests PRIVATE iiCameraIntegration::iiCameraIntegration)
        add_test(NAME iiCamera.color_decode COMMAND iiCameraColorDecodeTests "${CMAKE_CURRENT_BINARY_DIR}/color-decode")
        set_tests_properties(iiCamera.color_decode PROPERTIES TIMEOUT 30)
        find_program(IICAMERA_FFMPEG_EXECUTABLE ffmpeg)
        if(IICAMERA_FFMPEG_EXECUTABLE)
            add_executable(iiCameraRecognitionTests tests/recognition.cpp)
            target_link_libraries(iiCameraRecognitionTests PRIVATE iiCameraIntegration::iiCameraIntegration)
            add_test(NAME iiCamera.recognition_fixtures COMMAND "${CMAKE_COMMAND}"
                "-DFFMPEG=${IICAMERA_FFMPEG_EXECUTABLE}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/recognition"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/recognition_fixtures.cmake")
            set_tests_properties(iiCamera.recognition_fixtures PROPERTIES FIXTURES_SETUP recognitionFixture TIMEOUT 180)
            add_test(NAME iiCamera.recognition COMMAND iiCameraRecognitionTests "${CMAKE_CURRENT_BINARY_DIR}/recognition")
            set_tests_properties(iiCamera.recognition PROPERTIES FIXTURES_REQUIRED recognitionFixture TIMEOUT 60)
        endif()
    endif()
endif()
