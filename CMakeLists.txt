cmake_minimum_required(VERSION 3.24)
project(iiCameraIntergration VERSION 0.4.1 LANGUAGES CXX)
# File creation is owned by iiFileProvider.
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED HINTS "$ENV{IIFILEPROVIDER_PREFIX}" "$ENV{HOME}/.local/SDK/iiFileProvider")
link_libraries(iiFileProvider::Native)




set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(GenerateExportHeader)
include(CTest)
option(BUILD_SHARED_LIBS "Build a shared iiCameraIntergration library" ON)
option(IICAMERA_WITH_FFMPEG "Enable FFmpeg codec/format recognition and capture" ON)
option(IICAMERA_WITH_NATIVE "Enable the platform native camera metadata/control adapter" ON)
option(IICAMERA_WITH_IMAGE_PROCESSING "Enable optional iiGraphicAdjust photo processing component" OFF)
option(IICAMERA_BUILD_TOOLS "Build the camera/media inspection tool" ON)
if(PROJECT_IS_TOP_LEVEL AND CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    if(DEFINED ENV{HOME} AND NOT "$ENV{HOME}" STREQUAL "")
        set(CMAKE_INSTALL_PREFIX "$ENV{HOME}/.local/SDK/iiCameraIntergration" CACHE PATH "Install prefix" FORCE)
    elseif(DEFINED ENV{USERPROFILE} AND NOT "$ENV{USERPROFILE}" STREQUAL "")
        set(CMAKE_INSTALL_PREFIX "$ENV{USERPROFILE}/.local/SDK/iiCameraIntergration" CACHE PATH "Install prefix" FORCE)
    endif()
endif()
add_library(iiCameraIntergration src/iiCameraIntergration/Camera.cpp src/iiCameraIntergration/Devices.cpp
    src/iiCameraIntergration/Features.cpp)
add_library(iiCameraIntergration::iiCameraIntergration ALIAS iiCameraIntergration)
target_compile_features(iiCameraIntergration PUBLIC cxx_std_23)
target_include_directories(iiCameraIntergration PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
generate_export_header(iiCameraIntergration BASE_NAME IICAMERAINTERGRATION
    EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/include/iiCameraIntergration/Export.h")
if(NOT BUILD_SHARED_LIBS)
    target_compile_definitions(iiCameraIntergration PUBLIC IICAMERAINTERGRATION_STATIC_DEFINE)
endif()
set_target_properties(iiCameraIntergration PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    VERSION ${PROJECT_VERSION} SOVERSION ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR})
if(MSVC)
    target_compile_options(iiCameraIntergration PRIVATE /W4 /permissive-)
else()
    target_compile_options(iiCameraIntergration PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(iiCameraIntergration PRIVATE Threads::Threads)
if(IICAMERA_WITH_FFMPEG)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(IICameraFFmpeg REQUIRED IMPORTED_TARGET
        libavformat libavcodec libavdevice libavutil libswscale)
    target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/Media.cpp)
    target_link_libraries(iiCameraIntergration PRIVATE PkgConfig::IICameraFFmpeg)
else()
    target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/MediaUnavailable.cpp)
endif()
set(IICAMERA_HAS_APPLE_NATIVE OFF)
if(APPLE AND IICAMERA_WITH_NATIVE)
    enable_language(OBJCXX)
    set(IICAMERA_HAS_APPLE_NATIVE ON)
    target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/NativeCamera.mm)
    set_source_files_properties(src/iiCameraIntergration/NativeCamera.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    set_target_properties(iiCameraIntergration PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS OFF)
    find_library(IICameraAVFoundation AVFoundation REQUIRED)
    find_library(IICameraFoundation Foundation REQUIRED)
    find_library(IICameraCoreMedia CoreMedia REQUIRED)
    find_library(IICameraCoreVideo CoreVideo REQUIRED)
    target_link_libraries(iiCameraIntergration PRIVATE "${IICameraAVFoundation}" "${IICameraFoundation}" "${IICameraCoreMedia}" "${IICameraCoreVideo}")
elseif(ANDROID AND IICAMERA_WITH_NATIVE)
    if(CMAKE_SYSTEM_VERSION VERSION_LESS 24)
        message(STATUS "Native Camera2 unavailable before Android API 24; building the explicit fallback")
        target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/NativeCameraUnavailable.cpp)
    else()
        target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/NativeCameraAndroid.cpp)
        target_link_libraries(iiCameraIntergration PRIVATE camera2ndk)
    endif()
elseif(WIN32 AND IICAMERA_WITH_NATIVE)
    target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/NativeCameraWindows.cpp)
    target_link_libraries(iiCameraIntergration PRIVATE mfplat mf mfuuid strmiids ole32 oleaut32 propsys)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND IICAMERA_WITH_NATIVE)
    target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/NativeCameraLinux.cpp
        src/iiCameraIntergration/V4L2.cpp)
else()
    target_sources(iiCameraIntergration PRIVATE src/iiCameraIntergration/NativeCameraUnavailable.cpp)
endif()
if(IICAMERA_WITH_IMAGE_PROCESSING)
    find_package(iiGraphicAdjust 0.6 CONFIG REQUIRED)
    add_library(iiCameraImageProcessing src/iiCameraIntergration/ImageProcessing.cpp)
    add_library(iiCameraIntergration::ImageProcessing ALIAS iiCameraImageProcessing)
    target_compile_features(iiCameraImageProcessing PUBLIC cxx_std_23)
    target_link_libraries(iiCameraImageProcessing PUBLIC iiCameraIntergration::iiCameraIntergration iiGraphicAdjust::iiGraphicAdjust)
    target_include_directories(iiCameraImageProcessing PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
        $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    generate_export_header(iiCameraImageProcessing BASE_NAME IICAMERAIMAGEPROCESSING
        EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/include/iiCameraIntergration/ImageProcessingExport.h")
    if(NOT BUILD_SHARED_LIBS)
        target_compile_definitions(iiCameraImageProcessing PUBLIC IICAMERAIMAGEPROCESSING_STATIC_DEFINE)
    endif()
    set_target_properties(iiCameraImageProcessing PROPERTIES EXPORT_NAME ImageProcessing
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION}
        SOVERSION ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
        INSTALL_RPATH_USE_LINK_PATH TRUE)
    if(APPLE)
        set_target_properties(iiCameraImageProcessing PROPERTIES INSTALL_RPATH "@loader_path")
    elseif(UNIX)
        set_target_properties(iiCameraImageProcessing PROPERTIES INSTALL_RPATH "$ORIGIN")
    endif()
    if(MSVC)
        target_compile_options(iiCameraImageProcessing PRIVATE /W4 /permissive-)
    else()
        target_compile_options(iiCameraImageProcessing PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS iiCameraImageProcessing EXPORT iiCameraImageProcessingTargets
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(EXPORT iiCameraImageProcessingTargets NAMESPACE iiCameraIntergration::
        DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/iiCameraIntergration")
    install(FILES src/iiCameraIntergration/ImageProcessing.h
        "${CMAKE_CURRENT_BINARY_DIR}/include/iiCameraIntergration/ImageProcessingExport.h"
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiCameraIntergration)
endif()
set(_package_dir "${CMAKE_INSTALL_LIBDIR}/cmake/iiCameraIntergration")
install(TARGETS iiCameraIntergration EXPORT iiCameraIntergrationTargets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/iiCameraIntergration/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiCameraIntergration FILES_MATCHING PATTERN "*.h" PATTERN "ImageProcessing.h" EXCLUDE PATTERN "detail" EXCLUDE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/iiCameraIntergration/Export.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiCameraIntergration)
install(EXPORT iiCameraIntergrationTargets NAMESPACE iiCameraIntergration:: DESTINATION "${_package_dir}")
configure_package_config_file(cmake/iiCameraIntergrationConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntergrationConfig.cmake" INSTALL_DESTINATION "${_package_dir}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntergrationConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntergrationConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiCameraIntergrationConfigVersion.cmake" DESTINATION "${_package_dir}")
