cmake_minimum_required(VERSION 3.24)
project(iiCameraCorePackageCheck LANGUAGES CXX)
set(CMAKE_DISABLE_FIND_PACKAGE_iiGraphicAdjust TRUE)
if(MODE STREQUAL "Core")
    find_package(iiCameraIntegration 0.4 CONFIG REQUIRED COMPONENTS Core)
elseif(MODE STREQUAL "Optional")
    find_package(iiCameraIntegration 0.4 CONFIG REQUIRED COMPONENTS Core OPTIONAL_COMPONENTS ImageProcessing)
    if(iiCameraIntegration_ImageProcessing_FOUND OR TARGET iiCameraIntegration::ImageProcessing)
        message(FATAL_ERROR "Moved optional component was exposed by the camera SDK")
    endif()
elseif(MODE STREQUAL "Required")
    find_package(iiCameraIntegration 0.4 CONFIG REQUIRED COMPONENTS Core ImageProcessing)
else()
    message(FATAL_ERROR "Unknown camera package check")
endif()
if(NOT TARGET iiCameraIntegration::iiCameraIntegration)
    message(FATAL_ERROR "Camera core target is missing")
endif()
