cmake_minimum_required(VERSION 3.24)
# iisdk:consumer-application
project(iiCameraMediaConsumer LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
find_package(iiCameraIntegration 0.4 CONFIG REQUIRED COMPONENTS Core)
find_package(iiFileProviderNative 0.6 CONFIG REQUIRED HINTS "$ENV{IIFILEPROVIDER_PREFIX}" "$ENV{HOME}/.local/SDK/iiFileProvider")
add_executable(iiCameraInspect App/inspect.cpp)
target_compile_features(iiCameraInspect PRIVATE cxx_std_23)
target_link_libraries(iiCameraInspect PRIVATE iiCameraIntegration::iiCameraIntegration iiFileProvider::Native)
set_target_properties(iiCameraInspect PROPERTIES OUTPUT_NAME iicamera-inspect INSTALL_RPATH_USE_LINK_PATH TRUE)
if(APPLE)
    set_target_properties(iiCameraInspect PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
    set_target_properties(iiCameraInspect PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
install(TARGETS iiCameraInspect RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(BUILD_TESTING AND iiCameraIntegration_WITH_FFMPEG)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(IICAMERA_FFMPEG_EXECUTABLE ffmpeg REQUIRED)
    add_executable(iiCameraMediaTests tests/media.cpp)
    target_compile_features(iiCameraMediaTests PRIVATE cxx_std_23)
    target_include_directories(iiCameraMediaTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/App")
    target_link_libraries(iiCameraMediaTests PRIVATE iiCameraIntegration::iiCameraIntegration iiFileProvider::Native)
    add_test(NAME iiCamera.media COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/media_fixture.py"
        "${IICAMERA_FFMPEG_EXECUTABLE}" "$<TARGET_FILE:iiCameraMediaTests>" "${CMAKE_CURRENT_SOURCE_DIR}/resources")
    set_tests_properties(iiCamera.media PROPERTIES TIMEOUT 60 ENVIRONMENT "PYTHONPATH=${iiFileProviderNative_PYTHON_DIRECTORY}")
    add_test(NAME iiCamera.cli_help COMMAND iiCameraInspect --help)
    add_test(NAME iiCamera.cli_frame COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py"
        "$<TARGET_FILE:iiCameraInspect>" "${CMAKE_CURRENT_SOURCE_DIR}/resources")
    set_tests_properties(iiCamera.cli_frame PROPERTIES TIMEOUT 40 ENVIRONMENT "PYTHONPATH=${iiFileProviderNative_PYTHON_DIRECTORY}")
endif()
